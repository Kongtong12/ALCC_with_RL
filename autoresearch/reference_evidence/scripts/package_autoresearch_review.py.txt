"""Compact, evidence-complete private review handoff; sealed studies untouched."""
from pathlib import Path
import hashlib
import io
import json
import re
import shutil
import zipfile
import numpy as np

ROOT = Path(__file__).resolve().parents[1]
DEST = ROOT / 'deliverables/ICLR_AutoResearch_Review_20260916'
NAME = 'ICLR_AutoResearch_Review_20260916.zip'
DEST.mkdir(exist_ok=True)
files = {}


def digest(data):
    return hashlib.sha256(data).hexdigest()


def add(path, name=None):
    path = Path(path)
    files[name or path.relative_to(ROOT).as_posix()] = path.read_bytes()


previous = ROOT / 'artifacts/packages/ICLR_Warcraft_Results_Revision_20260916/ICLR_Warcraft_Results_Revision_20260916.zip'
with zipfile.ZipFile(previous) as archive:
    old_manifest = json.loads(archive.read('MANIFEST.json'))
    for name in archive.namelist():
        if name.startswith(('experiments/', 'evidence/', 'reviews/')) or (name.startswith('scripts/') and 'package_' not in name):
            content = archive.read(name)
            assert digest(content) == old_manifest[name]['sha256']
            files[name] = content

paper = ROOT / 'paper/latex/facepot'


def include_tex(path):
    name = path.relative_to(ROOT).as_posix()
    if name in files:
        return
    add(path)
    source = path.read_text(encoding='utf-8')
    for child in re.findall(r'\\input\{([^}]+)\}', source):
        include_tex(paper / (child if child.endswith('.tex') else child + '.tex'))
    for child in re.findall(r'\\includegraphics(?:\[[^]]*\])?\{([^}]+)\}', source):
        image = paper / child
        if not image.suffix:
            image = next(image.with_suffix(s) for s in ['.pdf', '.png', '.jpg'] if image.with_suffix(s).exists())
        add(image)
        for suffix in ['.svg', '.png']:
            variant = image.with_suffix(suffix)
            if variant.exists():
                add(variant)


include_tex(paper / 'main.tex')
for name in ['main.pdf', 'references.bib', 'iclr2027_conference.sty', 'iclr2027_conference.bst', 'natbib.sty', 'fancyhdr.sty']:
    add(paper / name)
for path in (ROOT / 'reviews/current/longrun_revision_20260916').glob('*'):
    if path.is_file():
        add(path)
for name in ['audit_longrun_review_package.py', 'package_autoresearch_review.py', 'generate_longrun_paper_assets.py',
             'verify_longrun_paper_revision.py', 'generate_fig1_scene.py']:
    add(ROOT / 'scripts' / name)
add(ROOT / 'reviews/current/PRO_RESEARCH_REVIEW_20260916.md', 'START_HERE_PRO_PROMPT_ZH.md')

studies = [('threshold_longrun', 'round0', 768), ('threshold_schedule_screen', 'work', 72),
           ('threshold_spo_screen', 'work', 108), ('threshold_smoothing_screen', 'work', 84),
           ('threshold_ema_screen', 'work', 36), ('threshold_meanblend_screen', 'work', 84)]
for study, work, expected in studies:
    folder = ROOT / 'experiments' / study
    w = folder / work
    freeze = json.loads((w / 'freeze.json').read_text())
    for relative, checksum in freeze['sources'].items():
        assert digest((folder / relative).read_bytes()) == checksum
        add(folder / relative)
    assert digest((w / 'protocol.json').read_bytes()) == freeze['protocol_sha256']
    for path in folder.iterdir():
        if path.is_file() and path.suffix in {'.py', '.md', '.json'}:
            add(path)
    for path in w.iterdir():
        if path.is_file() and (path.suffix in {'.json', '.csv', '.md', '.jsonl', '.log', '.sha256'} or '.before_' in path.name):
            add(path)
    receipts = list((w / 'runs').glob('*/*/result.json'))
    assert len(receipts) == expected, (study, len(receipts))
    for path in (w / 'runs').rglob('*.json'):
        add(path)
    for directory in ['failures', 'calibration', 'calibration.before_report_label']:
        for path in (w / directory).rglob('*.json'):
            add(path)
for path in (ROOT / 'experiments/threshold_longrun/research').iterdir():
    if path.is_file() and path.suffix in {'.py', '.md', '.json'}:
        add(path)

index = dict(scope='Lossless subset of original evaluation NPZ: selected_pred, costs, paths only. '
             'No probabilities, fixed-step predicted arrays, or neural checkpoints. '
             'Original NPZ hash links each subset to its archived receipt; that original NPZ is omitted.', runs=[])
w = ROOT / 'experiments/threshold_longrun/round0'
for folder in sorted((w / 'runs/confirmation').iterdir()):
    original = folder / 'evaluation.npz'
    record = json.loads((folder / 'evaluation.json').read_text())
    checksum = digest(original.read_bytes())
    assert checksum == record['arrays_sha256']
    with np.load(original, allow_pickle=False) as data:
        output = io.BytesIO()
        np.savez_compressed(output, **{key: data[key] for key in ['selected_pred', 'costs', 'paths']})
    name = f'evidence/longrun_selected_predictions/{folder.name}.npz'
    files[name] = output.getvalue()
    index['runs'].append(dict(id=folder.name, compact_path=name, compact_sha256=digest(files[name]),
                              original_evaluation_npz_sha256=checksum))
assert len(index['runs']) == 120
files['evidence/longrun_selected_predictions/INDEX.json'] = json.dumps(index, indent=2).encode()

files['MANIFEST.json'] = json.dumps({name: dict(bytes=len(data), sha256=digest(data)) for name, data in sorted(files.items())}, indent=2).encode()
destination = DEST / NAME
with zipfile.ZipFile(destination, 'w', zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
    for name, data in sorted(files.items()):
        archive.writestr(name, data)
with zipfile.ZipFile(destination) as archive:
    assert archive.testzip() is None
    for name, record in json.loads(archive.read('MANIFEST.json')).items():
        assert digest(archive.read(name)) == record['sha256']
    extracted = DEST / 'verified_extraction'
    version = 1
    while extracted.exists():
        version += 1
        extracted = DEST / f'verified_extraction_{version}'
    archive.extractall(extracted)
record = dict(status='CRC_MANIFEST_EXTRACTION_PASS', files=len(files), bytes=destination.stat().st_size,
              sha256=digest(destination.read_bytes()), selected_prediction_subsets=120, neural_checkpoints_included=False,
              scientific_receipts=1152, verified_extraction=extracted.name, previous_archive_sha256=digest(previous.read_bytes()))
(DEST / 'PACKAGE_VERIFICATION.json').write_text(json.dumps(record, indent=2), encoding='utf-8')
print(json.dumps(record, indent=2))
